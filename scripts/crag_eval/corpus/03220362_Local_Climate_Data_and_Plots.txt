			function main() {
				// Begin variables for the yearly climate form
				var date = new Date();
				var max = date.getFullYear();
				var month = date.getMonth();
				var eSID = document.getElementById("msid");
				var sid = eSID.options[eSID.selectedIndex].value;
				// If your sites' data begin in varying year use this approach for min.
				// Simply modify the sid and min year accordingly for your sites.
				if(sid == 'KCHS' || sid == 'KCXM' || sid == 'KSAV') {
					var min = 2014;
				} else if(sid == 'MNOP') {
					var min = 1999;
				} else {
					var min = 2010;
				}
				var selectYear = document.getElementById('myear');
				var selectMonth = document.getElementById('mmonth');
				// If selectYear already had options, remove them
				while(selectYear.hasChildNodes()) {
					selectYear.removeChild(selectYear.firstChild);
				}
				// Dynamically create the year menu using a min year and the current year
				for(var i = min; i <= max; i++) {
					var opt = document.createElement('option');
					opt.value = i;
					opt.innerHTML = i;
					selectYear.appendChild(opt);
				}
				// Default the year menu to select the current year
				selectYear.value = date.getFullYear();
				// Javascript defines the months as 0=Jan, 1=Feb, .. so add 1 to the month
				month = month + 1;
				if(month < 10) {
					// If the month is less than 10, add a leading 0.
					month = '0' + month;
				}
				// Default the month menu to select the current month
				selectMonth.value = month;
				// Call the showImage function so the images will be displayed
				showMonthlyData();
			}

			function loadXMLDoc(filename) {
				if(window.XMLHttpRequest) {
					xhttp = new XMLHttpRequest();
				} else // code for IE5 and IE6
				{
					xhttp = new ActiveXObject("Microsoft.XMLHTTP");
				}
				xhttp.open("GET", filename, false);
				xhttp.send();
				return xhttp.responseXML;
			}

			function getQueryVar(varName) {
				var queryStr = unescape(window.location.search) + '&';
				var regex = new RegExp('.*?[&\?]' + varName + '=(.*?)&.*');
				val = queryStr.replace(regex, "$1");
				return val == queryStr ? false : val;
			}
